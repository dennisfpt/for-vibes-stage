function generateTextPoints() { var offCanvas = document.createElement('canvas'); offCanvas.width = 1920; offCanvas.height = 1080; var offCtx = offCanvas.getContext('2d'); offCtx.fillStyle = '#fff'; offCtx.textAlign = 'center'; offCtx.textBaseline = 'middle'; // Đẩy cả 3 dòng chữ xuống thấp hẳn, nằm gọn phía dưới thanh logo đầu trang offCtx.font = '900 110px "Inter", sans-serif'; offCtx.fillText('FOR VIBES', 1920 / 2, 480); // Dòng 1 ở Y = 480 offCtx.font = '900 62px "Inter", sans-serif'; offCtx.fillText('THANK YOU FOR WATCHING', 1920 / 2, 620); // Dòng 2 ở Y = 620 offCtx.font = '900 62px "Inter", sans-serif'; offCtx.fillText('AND FEELING OUR STORY', 1920 / 2, 710); // Dòng 3 ở Y = 710 var imgData = offCtx.getImageData(0, 0, 1920, 1080).data; textTargetPoints = []; var step = 7; // Vùng quét pixel cũng được dời xuống thấp tương ứng (từ Y = 400 đến Y = 780) for (var y = 400; y < 780; y += step) { for (var x = 80; x < 1840; x += step) { var index = (y * 1920 + x) * 4; if (imgData[index + 3] > 128) { textTargetPoints.push({ x: x, y: y }); } } } }
