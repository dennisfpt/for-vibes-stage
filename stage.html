function generateTextPoints() { var offCanvas = document.createElement('canvas'); offCanvas.width = 1920; offCanvas.height = 1080; var offCtx = offCanvas.getContext('2d'); // Dòng 1: FOR VIBES (Giữ nguyên kích thước lớn) offCtx.fillStyle = '#fff'; offCtx.font = '900 140px "Inter", sans-serif'; offCtx.textAlign = 'center'; offCtx.textBaseline = 'middle'; offCtx.fillText('FOR VIBES', 1920 / 2, 240); // Dòng 2: THANK YOU FOR WATCHING AND FEELING OUR STORY // (Thu nhỏ font xuống 52px và chia thành 2 dòng phụ nếu cần, hoặc để chữ dài co giãn vừa vặn) offCtx.font = '900 48px "Inter", sans-serif'; offCtx.fillText('THANK YOU FOR WATCHING', 1920 / 2, 380); offCtx.fillText('AND FEELING OUR STORY', 1920 / 2, 445); var imgData = offCtx.getImageData(0, 0, 1920, 1080).data; textTargetPoints = []; var step = 8; // Mật độ hạt pháo hoa for (var y = 140; y < 500; y += step) { for (var x = 120; x < 1800; x += step) { var index = (y * 1920 + x) * 4; if (imgData[index + 3] > 128) { textTargetPoints.push({ x: x, y: y }); } } } }
